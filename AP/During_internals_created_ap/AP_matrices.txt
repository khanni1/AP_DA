import numpy as np

A = np.array([
    [1, 2],
    [3, 4]
])

B = np.array([
    [5, 6],
    [7, 8]
])

MATRIX OPERATIONS USING Numpy
A - B
A * B
A @ B
A.T
np.linalg.det(A)
np.linalg.inv(A)


np.sum(A)       # sum of all elements
np.max(A)       # maximum
np.min(A)       # minimum
np.mean(A)      # mean
np.diag(A)      # diagonal
